function g(r: Ok | Err | Empty, value: int | string) -> string {
  if let o: Ok = r {
    o.value
  } else if let e: Err = r {
    e.message
  } else if r is Empty {
    "empty"
  } else {
    "other"
  }
  if const narrowed: int = value {
    narrowed;
  };
  "done"
}
