1//! Core types used throughout the compiler.

3use std::fmt;
4
5use smol_str::SmolStr;
6use text_size::{TextRange, TextSize};
7
8/// Unique identifier for a source file
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
10pub struct FileId(u32);
11
12impl FileId {
13    pub fn new(id: u32) -> Self {
14        FileId(id)
15    }
16
17    pub fn as_u32(self) -> u32 {
18        self.0
19    }
20}
21
22impl fmt::Display for FileId {
23    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
24        write!(f, "FileId({})", self.0)
25    }
26}
27
28/// A span in source code, tracking both file and position
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
30pub struct Span {
31    pub file_id: FileId,
32    pub range: TextRange,
33}
34
35impl Span {
36    pub fn new(file_id: FileId, range: TextRange) -> Self {
37        Span { file_id, range }
38    }
39
40    pub fn at_offset(file_id: FileId, offset: TextSize) -> Self {
41        Span {
42            file_id,
43            range: TextRange::empty(offset),
44        }
45    }
46}

65/// Severity level for diagnostics
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub enum Severity {
68    Error,
69    Warning,
70    Info,
71}
72
73/// Base trait for all compiler diagnostics
74pub trait Diagnostic: std::fmt::Debug {
75    fn message(&self) -> String;
76    fn span(&self) -> Option<Span>;
77    fn severity(&self) -> Severity;
78}
