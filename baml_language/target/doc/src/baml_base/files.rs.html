1//! File management with Salsa 2022 API.
2//!
3//! Defines the core structures for accessing file contents and paths.
4
5use std::path::PathBuf;
6
7use crate::FileId;
8
9/// Input structure representing a source file in the compilation.
10///
11/// This is a salsa input, which means it's the primary way to provide
12/// source text to the compiler. The struct itself just stores an ID,
13/// with the actual data stored in the salsa database.
14#[salsa::input]
15pub struct SourceFile {
16    /// Source text for the file
17    #[returns(ref)]
18    pub text: String,
19
20    /// File path (for diagnostics and error reporting)
21    pub path: PathBuf,
22
23    /// The FileId associated with this source file
24    /// This is used to maintain compatibility with existing code
25    pub file_id: FileId,
26}
